# スタレ ガチャ計画プランナー 実装計画

> **実行するエージェントへ:** この計画は `superpowers:executing-plans` で、タスクを上から順に、バッチごとに人間のチェックポイントを挟んで実行する。サブエージェント駆動（subagent-driven-development）と Git worktree は使わない（`.claude/rules/開発の進め方.md`）。各ステップはチェックボックス（`- [ ]`）で進捗を追う。

**ゴール:** 「次のピックアップに、いまの貯蓄で間に合うか」に答える単体 HTML ツールを作り、GitHub Pages で公開する。

**構成:** `index.html` 1 ファイル。計算ロジックは `<script id="calc">` に純粋関数の集まり `Calc` として置き、画面の処理は `<script id="ui">` に置く。テストは Node.js の `node --test` で、`index.html` から `<script id="calc">` の中身だけを抜き出して読み込み、`Calc` の関数を直接検証する。ビルド手順はない。

**技術:** HTML / CSS / JavaScript（ライブラリなし）、Node.js v20（テストのみ）、Git、GitHub Pages。

**企画書:** `01_企画部/01_スタレ_ガチャ計画ツール/企画書_2026-09-12.md` — 計画はこの企画書を根拠にしている。実行者は両方を読む。

## 全体の制約（企画書より）

- HTML＋CSS＋JavaScript を 1 ファイル（`index.html`）に書く。外部ライブラリ、CDN、サーバーは使わない
- 公式の画像・ロゴ・音声は一切使わない。文字と CSS だけ
- フッターに「非公式のファンツールです。HoYoverse とは関係ありません」と作者名「ユニフ」を置く
- スマホ縦持ちの 1 カラムを第一に。PC でも崩れない
- 5★ 排出率: 1〜73 連目 0.6%、74 連目から 1 連ごとに 6% 加算、90 連目で 100%。すり抜け 50%
- 確定に必要な連数: 保証あり 90 − 天井、保証なし 180 − 天井
- 1 連 = 星玉 160。週は ÷7、バージョンは ÷42 で「1 日あたり」に換算
- 判定: 使える連数 ≥ 確定連数 → 余裕（緑）／ 確率 50% 以上 → ギリギリ（黄）／ 50% 未満 → 今は待て（赤）
- 「いまのガチャで使う予定の連数」が 1 以上なら、目標ガチャは天井 0・保証なしから計算し、その前提を画面に書く
- 収入の初期値は目安。企画書 3.2 の表の値を使い、「要調査」の 2 行は 0 を初期値にして「目安を自分で入れる」と表示する
- 入力は localStorage に自動保存。リセットボタン 1 つ
- 結果をコピー（判定・確率・根拠・不足・URL）
- ファイル名は英語（コード）。コミットメッセージは日本語で、末尾に `Co-Authored-By: Claude Opus 5 <noreply@anthropic.com>` を付ける

- [ ] **Step 3: `Calc` に実装する**

`index.html` の `<script id="calc">` を次に置き換える。

```html
<script id="calc">
const Calc = (() => {
  const JADE_PER_PULL = 160;
  const HARD_PITY = 90;
  const SOFT_PITY_START = 74;
  const BASE_RATE = 0.006;
  const SOFT_STEP = 0.06;
  const WIN_RATE = 0.5;

  // i 連目（1〜90）の 5★ 排出率
  function rateAt(i) {
    if (i >= HARD_PITY) return 1;
    if (i < SOFT_PITY_START) return BASE_RATE;
    return Math.min(1, BASE_RATE + SOFT_STEP * (i - (SOFT_PITY_START - 1)));
  }

  // pmf[k] = 天井 pity の状態から、いまから k 連目で初めて 5★ が出る確率。pmf[0] = 0
  function fiveStarPmf(pity) {
    const pmf = [0];
    let alive = 1; // まだ 5★ が出ていない確率
    for (let k = 1; pity + k <= HARD_PITY; k++) {
      const r = rateAt(pity + k);
      pmf.push(alive * r);
      alive *= 1 - r;
    }
    return pmf;
  }

  // n 連以内に 5★ が出る確率
  function probFiveStarWithin(pity, n) {
    const pmf = fiveStarPmf(pity);
    let s = 0;
    for (let k = 1; k <= n && k < pmf.length; k++) s += pmf[k];
    return Math.min(1, s);
  }

  // n 連以内に目標キャラ（ピックアップ）を引ける確率
  // 保証なし: 1 回目の 5★ で勝つ 50% ＋ すり抜けて残りの連数で 2 回目の 5★ を引く 50%
  function probLimitedWithin(state, n) {
    if (n <= 0) return 0;
    const pmf = fiveStarPmf(state.pity);
    let s = 0;
    for (let k = 1; k <= n && k < pmf.length; k++) {
      if (state.guaranteed) s += pmf[k];
      else s += pmf[k] * (WIN_RATE + (1 - WIN_RATE) * probFiveStarWithin(0, n - k));
    }
    return Math.min(1, s);
  }

  // 確定に必要な連数（最悪ケース）
  function pullsForGuarantee(state) {
    return (state.guaranteed ? HARD_PITY : HARD_PITY * 2) - state.pity;
  }

  return { JADE_PER_PULL, HARD_PITY, rateAt, fiveStarPmf, probFiveStarWithin, probLimitedWithin, pullsForGuarantee };
})();
window.Calc = Calc;
</script>
```

`<script id="calc">` の `pullsForGuarantee` の後（`return {...}` の前）に追加し、`return` の中身も差し替える。

```js
  const VERSION_DAYS = 42;
  const WEEK_DAYS = 7;
  const MS_PER_DAY = 86400000;

  // 空欄・NaN・負の値は 0 にそろえる
  function num(v) {
    const n = Number(v);
    return Number.isFinite(n) && n > 0 ? n : 0;
  }

  function toPulls(jade, tickets) {
    return Math.floor(num(jade) / JADE_PER_PULL) + Math.floor(num(tickets));
  }

  // 'YYYY-MM-DD' を UTC の日付にする（時差や夏時間で日数がずれないように）
  function parseISO(iso) {
    const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso || '');
    if (!m) return null;
    return Date.UTC(Number(m[1]), Number(m[2]) - 1, Number(m[3]));
  }

  function formatISO(ms) {
    return new Date(ms).toISOString().slice(0, 10);
  }

  function daysUntil(todayISO, targetISO) {
    const t = parseISO(todayISO);
    const g = parseISO(targetISO);
    if (t === null || g === null) return 0;
    return Math.max(0, Math.round((g - t) / MS_PER_DAY));
  }

  function addDays(iso, n) {
    return formatISO(parseISO(iso) + n * MS_PER_DAY);
  }

  // 収入の内訳を「1 日あたり」にそろえる
  function dailyIncome(inc) {
    const jadePerDay =
      num(inc.daily) +
      (inc.passOn ? num(inc.pass) : 0) +
      (inc.bpOn ? num(inc.bpJade) / VERSION_DAYS : 0) +
      num(inc.sim) / WEEK_DAYS +
      num(inc.endgame) / VERSION_DAYS +
      num(inc.version) / VERSION_DAYS;
    const ticketsPerDay = inc.bpOn ? num(inc.bpTickets) / VERSION_DAYS : 0;
    return { jadePerDay, ticketsPerDay };
  }

  return {
    JADE_PER_PULL, HARD_PITY,
    rateAt, fiveStarPmf, probFiveStarWithin, probLimitedWithin, pullsForGuarantee,
    num, toPulls, daysUntil, addDays, dailyIncome,
  };
```

```js
  const MAX_PITY = HARD_PITY - 1;

  function clamp(v, lo, hi) {
    return Math.min(hi, Math.max(lo, v));
  }

  function plan(inputs) {
    const daysLeft = daysUntil(inputs.today, inputs.targetDate);
    const inc = dailyIncome(inputs.income || {});
    const handPulls = toPulls(inputs.jade, inputs.tickets);
    // 680/42 × 42 のような割り算の往復で 23.9999… になるのを防ぐため、切り捨ての前にごく小さい値を足す
    const incomePulls = Math.floor((inc.jadePerDay * daysLeft) / JADE_PER_PULL + inc.ticketsPerDay * daysLeft + 1e-9);
    const plannedNow = Math.floor(num(inputs.plannedNow));
    const usablePulls = handPulls + incomePulls - plannedNow;

    // いまのガチャを引き切る前提なら、目標ガチャは天井 0・保証なしから始まる
    const pityReset = plannedNow >= 1;
    const pity = clamp(Math.floor(num(inputs.pity)), 0, MAX_PITY);
    const state = pityReset ? { pity: 0, guaranteed: false } : { pity, guaranteed: !!inputs.guaranteed };

    const guaranteeNeeded = pullsForGuarantee(state);
    const enough = usablePulls >= guaranteeNeeded;
    const probability = enough ? 1 : probLimitedWithin(state, usablePulls);
    const verdict = enough ? 'safe' : probability >= 0.5 ? 'close' : 'wait';

    const shortfallPulls = Math.max(0, guaranteeNeeded - usablePulls);
    const shortfallJade = shortfallPulls * JADE_PER_PULL;

    // 1 日あたり収入を連数にそろえ、不足分を貯めるのに何日かかるかを目標日から数える
    const pullsPerDay = inc.jadePerDay / JADE_PER_PULL + inc.ticketsPerDay;
    let waitDays = null;
    let waitUntil = null;
    if (shortfallPulls > 0 && pullsPerDay > 0) {
      waitDays = Math.ceil(shortfallPulls / pullsPerDay);
      waitUntil = addDays(inputs.today, daysLeft + waitDays);
    }

    const warnings = [];
    if (plannedNow > handPulls) warnings.push('planned_exceeds_hand');

    return {
      daysLeft, handPulls, incomePulls, plannedNow, usablePulls,
      state, pityReset, guaranteeNeeded, probability, verdict,
      shortfallPulls, shortfallJade, waitDays, waitUntil,
      jadePerDay: inc.jadePerDay + inc.ticketsPerDay * JADE_PER_PULL,
      warnings,
    };
  }

  return {
    JADE_PER_PULL, HARD_PITY,
    rateAt, fiveStarPmf, probFiveStarWithin, probLimitedWithin, pullsForGuarantee,
    num, toPulls, daysUntil, addDays, dailyIncome, plan,
  };
```

  const $ = (id) => document.getElementById(id);

    const d = new Date();
    const p = (n) => String(n).padStart(2, '0');
    return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
  }

  function readInputs() {
    return {
      today: todayISO(),
      targetDate: $('targetDate').value,
      jade: $('jade').value,
      tickets: $('tickets').value,
      pity: $('pity').value,
      guaranteed: $('guaranteed').checked,

        passOn: $('incPassOn').checked,
        pass: $('incPass').value,
        bpOn: $('incBpOn').checked,
        bpJade: $('incBpJade').value,
        bpTickets: $('incBpTickets').value,
        sim: $('incSim').value,

        version: $('incVersion').value,
      },

  }

  function fmtDate(iso) {
    const [y, m, d] = iso.split('-').map(Number);

    return n.toLocaleString('ja-JP');

```js
  const STORAGE_KEY = 'hsr-pull-planner:v1';
  const FIELDS = ['targetDate', 'jade', 'tickets', 'pity', 'plannedNow',
    'incDaily', 'incPass', 'incBpJade', 'incBpTickets', 'incSim', 'incEndgame', 'incVersion'];
  const CHECKS = ['guaranteed', 'incPassOn', 'incBpOn'];

  // 初期値を控えておき、リセット時に戻す
  const DEFAULTS = {};
  FIELDS.forEach((id) => { DEFAULTS[id] = $(id).value; });
  CHECKS.forEach((id) => { DEFAULTS[id] = $(id).checked; });

  function save() {
    try {
      const data = {};
      FIELDS.forEach((id) => { data[id] = $(id).value; });
      CHECKS.forEach((id) => { data[id] = $(id).checked; });
      localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
    } catch (e) { /* プライベートモードなどで保存できないときは黙って続ける */ }
  }

  function load() {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) return;
      const data = JSON.parse(raw);
      FIELDS.forEach((id) => { if (id in data) $(id).value = data[id]; });
      CHECKS.forEach((id) => { if (id in data) $(id).checked = !!data[id]; });
    } catch (e) { /* 壊れた保存データは無視する */ }
  }

  function reset() {
    FIELDS.forEach((id) => { $(id).value = DEFAULTS[id]; });
    CHECKS.forEach((id) => { $(id).checked = DEFAULTS[id]; });
    try { localStorage.removeItem(STORAGE_KEY); } catch (e) { /* 同上 */ }
    update();
  }

  $('form').addEventListener('input', () => { update(); save(); });
  $('reset').addEventListener('click', reset);
  load();
  update();

  window.App = { readInputs, render, update, save, load, reset };
```

（末尾の `window.App = ...` は Task 6 のものを置き換える。）

`update()` の定義を、最後の結果を控える形に変え、コピー用の関数を追加する。

```js
  let lastResult = null;

  function update() {
    const inputs = readInputs();
    renderIncomeSummary(inputs);
    lastResult = Calc.plan(inputs);
    render(lastResult);
  }

  function buildShareText(inputs, r) {
    const lines = [];
    lines.push(`【スタレ ガチャ計画】目標 ${inputs.targetDate ? fmtDate(inputs.targetDate) : '未設定'}`);
    lines.push(`判定: ${VERDICT_LABEL[r.verdict]} ／ 引ける確率 ${Math.round(r.probability * 100)}%`);
    lines.push(`手持ち ${r.handPulls}連 ＋ 見込み ${r.incomePulls}連 − 使う ${r.plannedNow}連 ＝ ${r.usablePulls}連（確定まで ${r.guaranteeNeeded}連）`);
    if (r.shortfallPulls > 0) {
      let s = `不足 ${r.shortfallPulls}連（星玉 ${r.shortfallJade}個）`;
      if (r.waitUntil) s += `。${fmtDate(r.waitUntil)} まで待てば確定`;
      lines.push(s);
    }
    lines.push(location.href.split('#')[0]);
    return lines.join('\n');
  }

  async function copyResult() {
    if (!lastResult) return;
    const text = buildShareText(readInputs(), lastResult);
    try {
      await navigator.clipboard.writeText(text);
    } catch (e) {
      // クリップボード API が使えない環境向けの逃げ道
      const ta = document.createElement('textarea');
      ta.value = text;
      document.body.appendChild(ta);
      ta.select();
      document.execCommand('copy');
      ta.remove();
    }
    const t = $('toast');
    t.hidden = false;
    setTimeout(() => { t.hidden = true; }, 2000);
  }

  $('copy').addEventListener('click', copyResult);
```

`$('copy').addEventListener(...)` は `$('reset').addEventListener('click', reset);` の隣に置く。`window.App` に `buildShareText` を足す。
